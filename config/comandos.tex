% 1. SISTEMA CENTRALIZADO DE TIPO DE DOCUMENTO (PROTOCOLO / TTI / TTII)

% Definición de booleanos de estado global
\newif\ifProtocolo
\newif\ifTTI
\newif\ifTTII

% Macro principal de configuración de tipo de documento
\newcommand{\TipoDocumento}[1]{%
  \edef\tipoDocParam{\detokenize{#1}}%
  \Protocolofalse
  \TTIfalse
  \TTIIfalse
  % Comparación normalizada
  \ifstrequal{#1}{PROTOCOLO}{%
    \Protocolotrue
    \def\TipoDocCodigo{PROTOCOLO}%
  }{%
  \ifstrequal{#1}{protocolo}{%
    \Protocolotrue
    \def\TipoDocCodigo{PROTOCOLO}%
  }{%
  \ifstrequal{#1}{TTI}{%
    \TTItrue
    \def\TipoDocCodigo{TTI}%
  }{%
  \ifstrequal{#1}{tti}{%
    \TTItrue
    \def\TipoDocCodigo{TTI}%
  }{%
  \ifstrequal{#1}{TT1}{%
    \TTItrue
    \def\TipoDocCodigo{TTI}%
  }{%
  \ifstrequal{#1}{TTII}{%
    \TTIItrue
    \def\TipoDocCodigo{TTII}%
  }{%
  \ifstrequal{#1}{ttii}{%
    \TTIItrue
    \def\TipoDocCodigo{TTII}%
  }{%
  \ifstrequal{#1}{TT2}{%
    \TTIItrue
    \def\TipoDocCodigo{TTII}%
  }{%
    \PackageWarning{PlantillaUPIIZ}{%
      Tipo de documento '#1' no reconocido. Opciones validas: PROTOCOLO, TTI, TTII. %
      Se usara TTI por defecto.%
    }%
    \TTItrue
    \def\TipoDocCodigo{TTI}%
  }}}}}}}}%
}

% ------------------------------------------------------------------------------
% 2. VALIDACIÓN DE CAMPOS OBLIGATORIOS Y NORMATIVA DE LÍNEA DE TRABAJO
% ------------------------------------------------------------------------------
\newcounter{totalAlumnosRegistrados}

\newcommand{\validarDatosDocumento}{%
  \ifcsvoid{tituloProyecto}{%
    \PackageWarning{PlantillaUPIIZ}{Falta definir \protect\tituloProyecto en config/datos.tex}%
  }{}%
  \ifcsvoid{alumnoANombre}{%
    \PackageWarning{PlantillaUPIIZ}{Falta definir \protect\alumnoANombre en config/datos.tex}%
  }{}%
  \ifcsvoid{asesorANombre}{%
    \PackageWarning{PlantillaUPIIZ}{Falta definir \protect\asesorANombre en config/datos.tex}%
  }{}%
  
  % Conteo de alumnos registrados
  \setcounter{totalAlumnosRegistrados}{0}%
  \ifcsvoid{alumnoANombre}{}{\stepcounter{totalAlumnosRegistrados}}%
  \ifcsvoid{alumnoBNombre}{}{\stepcounter{totalAlumnosRegistrados}}%
  \ifcsvoid{alumnoCNombre}{}{\stepcounter{totalAlumnosRegistrados}}%
  
  % Validación de Línea de Trabajo
  \ifundef{\NumeroLineaTrabajo}{%
    \PackageWarning{PlantillaUPIIZ}{Falta configurar \protect\LineaTrabajo en config/datos.tex (Opciones: I, II, III, IV, V, VI).}%
  }{%
    \ifLineaTrabajoValida
      \ifnumcomp{\value{totalAlumnosRegistrados}}{>}{\MaximoAlumnosLineaTrabajo}{%
        \PackageWarning{PlantillaUPIIZ}{%
          ADVERTENCIA: La Línea de Trabajo \NumeroLineaTrabajo\space admite un máximo de \MaximoAlumnosLineaTrabajo\space alumnos %
          conforme al Anexo 1 del Reglamento Interno de Trabajo Terminal. %
          Actualmente se tienen \arabic{totalAlumnosRegistrados} alumnos registrados.%
        }%
      }{}%
    \else
      \PackageWarning{PlantillaUPIIZ}{%
        Línea de trabajo no válida (\NumeroLineaTrabajo). Las opciones permitidas son: I, II, III, IV, V y VI.%
      }%
    \fi
  }%
  
  \ifProtocolo
    \ifcsvoid{areaUbicacion}{%
      \PackageWarning{PlantillaUPIIZ}{Para PROTOCOLO se sugiere definir \protect\areaUbicacion en config/datos.tex}%
    }{}%
    \ifcsvoid{intencionTitulacion}{%
      \PackageWarning{PlantillaUPIIZ}{Para PROTOCOLO se sugiere definir \protect\intencionTitulacion en config/datos.tex}%
    }{}%
  \fi
}

% Imprime los autores en la Portada Principal
\newcommand{\imprimirAutoresPortada}{%
  \begin{minipage}[t]{0.9\textwidth}
    \centering\fontsize{13}{16}\selectfont
    \textbf{Presenta(n):}\\[0.5em]
    \alumnoANombre%
    \ifcsvoid{alumnoBNombre}{}{%
      \\[0.3em]\alumnoBNombre%
    }%
    \ifcsvoid{alumnoCNombre}{}{%
      \\[0.3em]\alumnoCNombre%
    }%
  \end{minipage}%
}

% Imprime los asesores en la Portada Principal
\newcommand{\imprimirAsesoresPortada}{%
  \begin{minipage}[t]{0.9\textwidth}
    \centering\fontsize{13}{16}\selectfont
    \textbf{Dirección / Asesoría:}\\[0.5em]
    \asesorANombre%
    \ifcsvoid{asesorBNombre}{}{%
      \\[0.3em]\asesorBNombre%
    }%
    \ifcsvoid{asesorCNombre}{}{%
      \\[0.3em]\asesorCNombre%
    }%
  \end{minipage}%
}

% Línea para firmas en la Portada Interna
\newcommand{\lineaFirma}[2]{%
  \begin{minipage}[t]{0.45\textwidth}
    \centering
    \vspace{1.0cm}
    \underline{\hspace{6.0cm}}\\[0.3em]
    {\fontsize{10}{12}\bfseries\sffamily #1}\\[0.2em]
    {\fontsize{9}{11}\sffamily #2}
  \end{minipage}%
}
